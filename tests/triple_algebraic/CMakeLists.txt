cmake_minimum_required(VERSION 3.16)
project(triple_algebraic_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
foreach(component interval weighted physical promotion geometry angular radius panel events integrator root_pair critical_pair mapped_limb gradient_replay gradient_hermite fused_transport first_order frame reference surface special resultant point_sign lens_coefficients magnitude)
    add_executable(triple_${component} test_${component}.cpp)
    target_compile_features(triple_${component} PRIVATE cxx_std_17)
    target_include_directories(triple_${component} PRIVATE ../../src)
    target_compile_options(triple_${component} PRIVATE
        -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    target_link_libraries(triple_${component} PRIVATE quadmath Threads::Threads)
    add_test(NAME triple_${component} COMMAND triple_${component})
endforeach()

option(LCBININT_TRIPLE_STANDALONE_PYTHON "Build isolated triple Python binding" OFF)
if(LCBININT_TRIPLE_STANDALONE_PYTHON)
    find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
    execute_process(COMMAND ${Python3_EXECUTABLE} -m pybind11 --cmakedir
        OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE pybind11_result)
    if(NOT pybind11_result EQUAL 0)
        message(FATAL_ERROR "Cannot locate pybind11 CMake package")
    endif()
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_triple_algebraic_test ../../python/bind_triple_algebraic.cpp)
    target_compile_definitions(_triple_algebraic_test PRIVATE LCBININT_TRIPLE_STANDALONE_MODULE=1)
    target_include_directories(_triple_algebraic_test PRIVATE ../../src)
    target_compile_features(_triple_algebraic_test PRIVATE cxx_std_17)
    target_compile_options(_triple_algebraic_test PRIVATE -fno-fast-math -ffp-contract=off)
    target_link_libraries(_triple_algebraic_test PRIVATE quadmath Threads::Threads)
endif()
